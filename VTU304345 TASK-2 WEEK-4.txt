t = int(input())

for _ in range(t):
    s = input().strip()

    n = len(s)
    mid = n // 2

    first = s[:mid]

    if n % 2 == 0:
        second = s[mid:]
    else:
        second = s[mid + 1:]

    if sorted(first) == sorted(second):
        print("YES")
    else:
        print("NO")
